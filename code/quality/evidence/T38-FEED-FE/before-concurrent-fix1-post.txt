import {useEffect,useRef,useState} from 'react';import {Link,useNavigate} from 'react-router-dom';import {Heart,MessageSquare,MessageCircle,Flag,Pencil,Trash2} from 'lucide-react';import type {FeedPostDTO} from '@fg/contracts';import {GameLogo} from '../../components/ui/GameLogo.js';import {FeedClient} from './api.js';import {FeedComposer,feedTypes} from './FeedComposer.js';import {FeedComments} from './FeedComments.js';import {ActionDialog} from '../moderation/ActionDialog.js';import {ModalFrame} from '../moderation/ModalFrame.js';import {ReportDialog} from '../moderation/ReportDialog.js';import {ActionApiError,actionError} from '../moderation/request.js';import {createDirectConversation} from '../direct-messages/api.js';
const client=new FeedClient();
export function FeedPost({post,actorId,canWrite,admin,detail=false,onChange,onDeleted,onDenied,onReload}:{post:FeedPostDTO;actorId?:string;canWrite:boolean;admin:boolean;detail?:boolean;onChange:(post:FeedPostDTO)=>void;onDeleted:()=>void;onDenied:(error:unknown)=>void;onReload:()=>void}){
 const [busy,setBusy]=useState(false),[error,setError]=useState(''),[comments,setComments]=useState(detail),[edit,setEdit]=useState(false),[remove,setRemove]=useState(false),[report,setReport]=useState(false),[preview,setPreview]=useState<number|null>(null),pending=useRef(false),controller=useRef<AbortController|null>(null),alive=useRef(true),navigate=useNavigate();
 useEffect(()=>{alive.current=true;return()=>{alive.current=false;controller.current?.abort();};},[]);
 function fail(problem:unknown){if(!alive.current)return;setError(actionError(problem));if(problem instanceof ActionApiError&&[401,403,404].includes(problem.status))onDenied(problem);}
 async function like(){if(!canWrite||pending.current)return;pending.current=true;setBusy(true);setError('');const abort=new AbortController();controller.current=abort;try{const row=await client.like(post.id,!post.likedByMe,abort.signal);if(!abort.signal.aborted)onChange({...post,likeCount:row.likeCount,likedByMe:row.likedByMe});}catch(problem){if(!abort.signal.aborted)fail(problem);}finally{pending.current=false;if(!abort.signal.aborted)setBusy(false);}}
 async function message(){if(pending.current||!canWrite)return;pending.current=true;setBusy(true);setError('');try{const row=await createDirectConversation(post.author.id);if(alive.current)navigate('/?dm='+encodeURIComponent(row.conversationId));}catch(problem){if(alive.current)setError(actionError(problem));}finally{pending.current=false;if(alive.current)setBusy(false);}}
 return <article className="fg-feed-post" data-post-id={post.id} aria-label={'Bài viết của '+post.author.displayName}><header><Link className="fg-feed-avatar" to={'/users/'+encodeURIComponent(post.author.id)} aria-label={'Hồ sơ '+post.author.displayName}>{post.author.avatarUrl?<img src={post.author.avatarUrl} alt=""/>:<span aria-hidden="true">{post.author.displayName.slice(0,1)}</span>}</Link><div className="fg-feed-author"><Link to={'/users/'+encodeURIComponent(post.author.id)}>{post.author.displayName}</Link><Link to={'/feed/posts/'+encodeURIComponent(post.id)}><time dateTime={post.createdAt}>{new Date(post.createdAt).toLocaleString('vi-VN')}</time></Link><div className="fg-feed-badges"><span><GameLogo game={post.game.title} decorative/>{post.game.title}</span><span className="fg-feed-type">{feedTypes[post.type]}</span></div></div>{canWrite&&actorId!==post.author.id&&<button type="button" className="fg-feed-dm" disabled={busy} onClick={()=>void message()} aria-label={'Nhắn riêng '+post.author.displayName}><MessageCircle size={16} aria-hidden="true"/><span>Nhắn riêng</span></button>}</header>
  {edit?<FeedComposer initial={post} onCancel={()=>setEdit(false)} onSaved={row=>{onChange(row);setEdit(false);}}/>:<><p className="fg-feed-text fg-feed-post-body">{post.body}</p><div className="fg-feed-images">{post.images.map((image,i)=><button type="button" key={image.id} onClick={()=>setPreview(i)} aria-label={'Xem ảnh bài viết '+(i+1)}><img src={image.url} alt={'Ảnh bài viết '+(i+1)} loading="lazy"/></button>)}</div></>}
  <div className="fg-feed-post-actions">{canWrite?<button type="button" aria-label={post.likedByMe?'Bỏ thích bài viết':'Thích bài viết'} aria-pressed={post.likedByMe} disabled={busy||edit} onClick={()=>void like()}><Heart size={18} aria-hidden="true"/>{post.likedByMe?'Đã thích':'Thích'}</button>:<Link to="/login">Đăng nhập để thích</Link>}<button type="button" aria-expanded={comments} onClick={()=>setComments(v=>!v)}><MessageSquare size={18} aria-hidden="true"/>Bình luận</button><span>{post.likeCount} lượt thích</span><span>{post.commentCount} bình luận</span></div>
  {canWrite&&<div className="fg-feed-manage">{actorId===post.author.id&&<button type="button" aria-label="Sửa bài viết" disabled={busy||edit} onClick={()=>setEdit(true)}><Pencil size={15} aria-hidden="true"/>Sửa</button>}{(actorId===post.author.id||admin)&&<button type="button" aria-label="Xóa bài viết" disabled={busy||edit} onClick={()=>setRemove(true)}><Trash2 size={15} aria-hidden="true"/>Xóa</button>}<button type="button" aria-label="Báo cáo bài viết" disabled={busy||edit} onClick={()=>setReport(true)}><Flag size={15} aria-hidden="true"/>Báo cáo</button></div>}
  {error&&<p role="alert" className="fg-feed-error">{error}</p>}{comments&&<FeedComments postId={post.id} actorId={actorId} canWrite={canWrite} admin={admin} onDenied={onDenied} onChanged={onReload}/>}
  {remove&&<ActionDialog title="Xóa bài viết" message="Xác nhận xóa bài cùng quyền truy cập bình luận và ảnh của bài. Thao tác này không thể hoàn tác trong giao diện." confirmLabel="Xác nhận xóa bài viết" onClose={()=>setRemove(false)} onConfirm={async(reason,signal)=>{try{await client.remove(post.id,reason,signal);}catch(problem){if(!signal.aborted)fail(problem);throw problem;}}} onSuccess={onDeleted}/>}
  {report&&<ReportDialog target={{targetType:'FEED_POST',targetId:post.id,label:'Bài viết của '+post.author.displayName}} onClose={()=>setReport(false)}/>}
  {preview!==null&&post.images[preview]&&<ModalFrame title="Ảnh bài viết" onClose={()=>setPreview(null)}><img className="fg-feed-preview" src={post.images[preview].url} alt={'Ảnh bài viết '+(preview+1)}/></ModalFrame>}
 </article>;
}
