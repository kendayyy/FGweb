    const owner=await pageFor('owner'),other=await pageFor('other'),admin=await pageFor('admin');
    phase='owner-create';current=owner;await owner.goto(origin+'/support');await owner.getByRole('button',{name:'Yêu cầu mới'}).click();await owner.getByRole('button',{name:'Gửi yêu cầu',exact:true}).click();await expect(owner.getByRole('alert')).toContainText('nội dung');await owner.getByLabel('Nội dung hỗ trợ',{exact:true}).fill('T38_SYNTHETIC_OWNER_REQUEST');await owner.getByRole('button',{name:'Gửi yêu cầu',exact:true}).click();await expect(owner.getByLabel('Lịch sử hỗ trợ').getByText('T38_SYNTHETIC_OWNER_REQUEST',{exact:true})).toBeVisible();await expect(owner.getByLabel('Lịch sử hỗ trợ')).toContainText('FG đã nhận tin nhắn của bạn.');await capture(owner,'Người dùng gửi yêu cầu thật','Form trống bị chặn; nội dung và xác nhận chỉ xuất hiện sau khi lưu vào PostgreSQL.');

    phase='owner-realtime';current=owner;await owner.bringToFront();await expect(owner.getByLabel('Lịch sử hỗ trợ').getByText('T38_SYNTHETIC_ADMIN_REPLY',{exact:true})).toBeVisible();await expect(owner.getByRole('button',{name:'Mở yêu cầu Chat hỗ trợ'})).toContainText('Đang xử lý');await expect(owner.getByText('T38_SYNTHETIC_INTERNAL_NOTE',{exact:true})).toHaveCount(0);const userDetail=await owner.request.get(origin+'/api/v1/support/requests/'+requestId);expect(userDetail.status()).toBe(200);expect(JSON.stringify(await userDetail.json())).not.toContain('T38_SYNTHETIC_INTERNAL_NOTE');await owner.getByRole('button',{name:'Đánh dấu đã đọc'}).click();await capture(owner,'Người gửi nhận phản hồi','Socket/REST đồng bộ phản hồi và trạng thái thật; notes không có trong DOM hoặc API người gửi.');checks.push('actual owner realtime canonical state and note isolation');

    phase='report-create';await owner.goto(origin+'/users/'+actors.other.id);await expect(owner.getByText('Chặn sẽ ẩn tin Lobby của người này và ngăn DM, lời mời mới.     phase='report-process';current=admin;await admin.goto(origin+'/admin/reports');await admin.getByRole('button',{name:'Mở báo cáo '+reports[0].id}).click();await admin.getByRole('button',{name:'Đánh dấu đã xử lý'}).click();await expect(admin.getByLabel('Báo cáo được chọn')).toContainText('Đã xử lý');expect((await pool.query('SELECT status FROM reports WHERE id=$1',[reports[0].id])).rows[0].status).toBe('RESOLVED');await capture(admin,'Xử lý báo cáo thật','Admin xử lý; DOM và trạng thái PostgreSQL đều xác nhận RESOLVED.');checks.push('actual user report and persisted admin resolution');
    phase='visual';await admin.goto(origin+'/admin/support/'+requestId);
    for(const viewport of [{width:1440,height:900},{width:1280,height:800},{width:390,height:844},{width:360,height:800}]){
      await admin.setViewportSize(viewport);for(const theme of ['light','dark']){const currentTheme=await admin.locator('html').getAttribute('data-theme');if(currentTheme!==theme)await admin.getByRole('button',{name:theme==='light'?'Chuyển sang Sáng':'Chuyển sang Tối',exact:true}).click();await expect(admin.locator('html')).toHaveAttribute('data-theme',theme);await expect(admin.getByRole('heading',{name:'Góp ý & Hỗ trợ',exact:true})).toBeVisible();if(viewport.width<=760){const row=admin.getByRole('button',{name:'Mở yêu cầu Chat hỗ trợ'});if(await row.isVisible())await row.click();await expect(admin.getByRole('button',{name:'Chi tiết yêu cầu'})).toBeVisible();await admin.getByRole('button',{name:'Chi tiết yêu cầu'}).click();await expect(admin.getByLabel('Nội dung ghi chú nội bộ')).toBeVisible();await admin.getByRole('button',{name:'Cuộc hỗ trợ',exact:true}).click();}expect(await admin.evaluate(()=>document.documentElement.scrollWidth<=innerWidth+1)).toBe(true);await capture(admin,'FG '+viewport.width+'×'+viewport.height+' '+theme,'Kiểm tra viewport thật, bố cục hỗ trợ và chuyển pane trên điện thoại; không tràn ngang.');}
    }

xác nhận có lý do thu hồi phiên thật.');checks.push('actual cancel/no-write/focus return and reasoned BAN session revocation');await peerRow.getByRole('button',{name:'Mở khóa tài khoản FG Test Peer'}).click();dialog=admin.getByRole('dialog',{name:'Mở khóa tài khoản'});await dialog.getByLabel('Lý do xử lý').fill('T38_SYNTHETIC_UNBAN_REASON');await dialog.getByRole('button',{name:'Xác nhận mở khóa',exact:true}).click();await expect(peerRow.getByRole('button',{name:'Khóa tài khoản FG Test Peer'})).toBeVisible();expect((await pool.query('SELECT id FROM session WHERE user_id=$1',[actors.other.id])).rowCount).toBe(0);expect((await other.request.get(origin+'/api/v1/me')).status()).toBe(401);await capture(admin,'Mở khóa không phục hồi phiên cũ','Mở khóa thật và list cập nhật; cookie cũ vẫn bị từ chối, người dùng phải đăng nhập lại.');checks.push('actual UNBAN reflects list and never restores stale session');
    phase='current-role';await admin.goto(origin+'/admin/support/'+requestId);await expect(admin.locator('.fg-support-internal').getByText('T38_SYNTHETIC_INTERNAL_NOTE',{exact:true})).toBeVisible();await pool.query('UPDATE "user" SET name=$2 WHERE id=$1',[actors.admin.id,'former-admin']);await expect(admin.getByRole('heading',{name:'Không có quyền quản trị'})).toBeVisible();await expect(admin.getByText('T38_SYNTHETIC_INTERNAL_NOTE',{exact:true})).toHaveCount(0);await capture(admin,'Thu hồi quyền quản trị hiện hành','Đổi quyền persisted làm socket thu hồi; notes và trang bảo vệ được xóa ngay.');checks.push('actual current sole-admin namespace revocation clears private notes');

    fs.writeFileSync(dir+'/result.json',JSON.stringify({runId,startedAt,completedAt:new Date().toISOString(),checks,captureIds:captures,actualHeadedChrome:true,chromeVersion:browser.version(),actualPostgresRedis:true,syntheticFixturesOnly:true,actualKingameProvisioned:false,taskAcceptancePending:true},null,2)+'\n',{flag:'wx'});
