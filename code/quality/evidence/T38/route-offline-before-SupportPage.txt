import {useCallback,useEffect,useMemo,useRef,useState} from 'react';import {Link,useParams} from 'react-router-dom';import {LifeBuoy,Plus,Search,ArrowLeft} from 'lucide-react';import {SupportCreateRequestSchema,SupportKindSchema,SupportStatusSchema,type SupportKind,type SupportStatus,type SupportUserRequest} from '@fg/contracts';import {useAuth} from '../auth/index.js';import {registerAuthCleanup} from '../auth/lifecycle.js';import {useAdminAccess} from '../admin/AdminAccess.js';import {ActionApiError,actionError} from '../moderation/request.js';import {Button} from '../../components/ui/Button.js';import {InitialLoading} from '../../components/ui/InitialLoading.js';import {SupportClient} from './api.js';import {SupportThread} from './SupportThread.js';import {type SupportDraft} from './drafts.js';import './support.css';
import {statusLabel,kindLabel,supportTime,type SupportRequest} from './labels.js';
function useCompact(){const [compact,setCompact]=useState(()=>window.matchMedia?.('(max-width:760px)').matches??false);useEffect(()=>{const media=window.matchMedia?.('(max-width:760px)');if(!media)return;const changed=()=>setCompact(media.matches);media.addEventListener('change',changed);return()=>media.removeEventListener('change',changed);},[]);return compact;}
export function SupportPage({admin=false}:{admin?:boolean}){const auth=useAuth(),access=useAdminAccess(),params=useParams<{requestId?:string}>();if(auth.isLoading)return <InitialLoading pending layout="bootstrap" label="Đang kiểm tra phiên hỗ trợ"/>;if(!auth.user||auth.isError||auth.isSigningOut)return <section className="fg-support-page fg-support-guest"><h1>Góp ý & Hỗ trợ</h1><p>Đăng nhập để gửi yêu cầu và xem lịch sử hỗ trợ riêng của bạn.</p><Link className="fg-btn fg-btn-primary" to="/login">Đăng nhập</Link></section>;if(admin&&(!access.allowed||access.actorId!==auth.user.id))return null;return <OwnedSupport key={auth.user.id+':'+admin} actorId={auth.user.id} admin={admin} initialId={params.requestId}/>;}
function OwnedSupport({actorId,admin,initialId}:{actorId:string;admin:boolean;initialId?:string}){
 const auth=useAuth(),access=useAdminAccess(),client=useMemo(()=>new SupportClient(actorId,admin),[actorId,admin]),compact=useCompact(),vault=useRef(new Map<string,SupportDraft>()),active=useRef(true),controllers=useRef(new Set<AbortController>()),latest=useRef<AbortController|null>(null),seenCursors=useRef(new Set<string>());
 const [items,setItems]=useState<SupportRequest[]>([]),[cursor,setCursor]=useState<string|null>(null),[loading,setLoading]=useState(true),[more,setMore]=useState(false),[error,setError]=useState(''),[blocked,setBlocked]=useState(false),[selected,setSelected]=useState<string|null>(initialId??null),[creating,setCreating]=useState(false),[pane,setPane]=useState<'list'|'thread'|'detail'>(initialId?'thread':'list'),[status,setStatus]=useState<SupportStatus|''>(''),[kind,setKind]=useState<SupportKind|''>(''),[search,setSearch]=useState(''),[unreadOnly,setUnreadOnly]=useState(false);
 const deny=useCallback((problem:unknown)=>{if(!active.current)return;if(problem instanceof ActionApiError&&[401,403].includes(problem.status)){setBlocked(true);setItems([]);setSelected(null);vault.current.clear();for(const controller of controllers.current)controller.abort();if(admin)access.revoke();if(problem.status===401)void auth.refetchUser().catch(()=>{});}setError(actionError(problem));},[admin,access.revoke,auth.refetchUser]);
 const refresh=useCallback(async(next?:string)=>{if(!active.current||blocked)return;if(next){if(seenCursors.current.has(next))return;setMore(true);}else{latest.current?.abort();setLoading(true);seenCursors.current.clear();}const controller=new AbortController();controllers.current.add(controller);if(!next)latest.current=controller;try{const page=await client.list(controller.signal,{limit:30,...(kind?{kind}:{}),...(status?{status}:{}),...(next?{cursor:next}:{})});if(controller.signal.aborted||!active.current)return;if(next){seenCursors.current.add(next);setItems(previous=>{const map=new Map(previous.map(row=>[row.id,row]));for(const row of page.items)map.set(row.id,row);return[...map.values()];});}else setItems(page.items);setCursor(page.nextCursor);setError('');}catch(problem){if(!controller.signal.aborted&&active.current)deny(problem);}finally{controllers.current.delete(controller);if(!controller.signal.aborted&&active.current){setLoading(false);setMore(false);}}},[client,kind,status,blocked,deny]);
 useEffect(()=>{active.current=true;void refresh();return()=>{active.current=false;for(const controller of controllers.current)controller.abort();};},[refresh]);
 const refreshRef=useRef(refresh);refreshRef.current=refresh;
 useEffect(()=>{const clear=()=>{setBlocked(true);setItems([]);setSelected(null);setCreating(false);vault.current.clear();for(const controller of controllers.current)controller.abort();};const unregister=registerAuthCleanup(clear);const timer=setInterval(()=>{if(!document.hidden)void refreshRef.current();},15000);return()=>{unregister();clearInterval(timer);vault.current.clear();};},[actorId,admin]);
 const choose=(id:string)=>{setSelected(id);setCreating(false);setPane('thread');};
 const updated=useCallback((row:SupportRequest)=>{if(active.current)setItems(previous=>previous.some(item=>item.id===row.id)?previous.map(item=>item.id===row.id?row:item):[row,...previous]);},[]);
 const visible=items.filter(row=>(!unreadOnly||row.unreadCount>0)&&(!search||[row.title,'owner'in row?row.owner.displayName:''].some(text=>text.toLocaleLowerCase('vi').includes(search.toLocaleLowerCase('vi')))));
 if(blocked)return <section className="fg-support-page"><h1>Góp ý & Hỗ trợ</h1><p role="alert">{error||'Phiên hỗ trợ không còn khả dụng. Nội dung riêng đã được xóa.'}</p><Link to="/login" className="fg-btn fg-btn-primary">Đăng nhập lại</Link></section>;
 return <section className={'fg-support-page '+(admin?'is-admin':'')}><header className="fg-support-heading"><div><h1>Góp ý & Hỗ trợ</h1><p>{admin?'Tiếp nhận và phản hồi cộng đồng':'Trao đổi riêng với quản trị FG'}</p></div>{!admin&&<Button onClick={()=>{setCreating(true);setSelected(null);setPane('thread');}} leftIcon={<Plus size={18} aria-hidden="true"/>}>Yêu cầu mới</Button>}</header>
  <div className="fg-support-workspace">
   {(!compact||pane==='list')&&<aside className="fg-support-inbox" aria-label="Hộp thư hỗ trợ"><header><h2>{admin?'Hộp thư':'Yêu cầu của bạn'}</h2><Button variant="ghost" onClick={()=>void refresh()} disabled={loading}>Tải lại</Button></header><label className="fg-support-search"><Search size={18} aria-hidden="true"/><span className="fg-sr-only">Tìm trong các yêu cầu đã tải</span><input aria-label="Tìm trong các yêu cầu đã tải" placeholder="Tìm trong danh sách đã tải…" value={search} onChange={event=>setSearch(event.target.value)}/></label><div className="fg-support-filters"><label>Trạng thái<select value={status} onChange={event=>{setStatus(event.target.value as SupportStatus|'');setSelected(null);setPane('list');}}><option value="">Tất cả</option>{SupportStatusSchema.options.map(value=><option key={value} value={value}>{statusLabel[value]}</option>)}</select></label><label>Loại yêu cầu<select value={kind} onChange={event=>{setKind(event.target.value as SupportKind|'');setSelected(null);setPane('list');}}><option value="">Tất cả</option>{SupportKindSchema.options.map(value=><option key={value} value={value}>{kindLabel[value]}</option>)}</select></label></div><label className="fg-support-check"><input type="checkbox" checked={unreadOnly} onChange={event=>setUnreadOnly(event.target.checked)}/>Chỉ yêu cầu chưa đọc trong danh sách</label>
   {loading&&items.length===0?<p role="status">Đang tải yêu cầu…</p>:visible.length===0?<p className="fg-support-empty">{items.length?'Không có yêu cầu khớp bộ lọc.':admin?'Chưa có yêu cầu hỗ trợ.':'Bạn chưa có yêu cầu hỗ trợ.'}</p>:<ul>{visible.map(row=><li key={row.id}><button type="button" aria-label={'Mở yêu cầu '+row.title} aria-current={selected===row.id?'true':undefined} onClick={()=>choose(row.id)}><span className="fg-support-avatar" aria-hidden="true">{'owner'in row?row.owner.displayName.slice(0,1):<LifeBuoy size={20}/>}</span><span className="fg-support-item"><strong>{'owner'in row?row.owner.displayName:row.title}</strong>{'owner'in row&&<span>{row.title}</span>}<span className={'fg-support-status '+row.status}>{statusLabel[row.status]}</span><time dateTime={row.lastActivityAt}>{supportTime(row.lastActivityAt)}</time></span>{row.unreadCount>0&&<span className="fg-support-unread" aria-label={row.unreadCount+' tin chưa đọc'}>{row.unreadCount}</span>}</button></li>)}</ul>}{cursor&&<Button variant="outline" disabled={more||loading} onClick={()=>void refresh(cursor)}>{more?'Đang tải…':'Tải thêm yêu cầu'}</Button>}{error&&<p role="alert" className="fg-action-error">{error}</p>}</aside>}
   {creating?<div className="fg-support-thread" hidden={compact&&pane==='list'}>{compact&&<button type="button" className="fg-btn fg-btn-ghost" onClick={()=>setPane('list')}><ArrowLeft size={18} aria-hidden="true"/>Hộp thư</button>}<NewSupport client={client} onCreated={row=>{updated(row);choose(row.id);}} onCancel={()=>{setCreating(false);setPane('list');}} onDenied={deny}/></div>:selected?<SupportThread key={selected} id={selected} client={client} vault={vault.current} showThread={!compact||pane==='thread'} showDetails={admin&&(!compact||pane==='detail')} compact={compact} onBack={()=>setPane('list')} onDetails={()=>setPane('detail')} onThread={()=>setPane('thread')} onUpdated={updated} onDenied={deny}/>:!compact&&<div className="fg-support-placeholder"><LifeBuoy size={32} aria-hidden="true"/><h2>{admin?'Chọn yêu cầu để hỗ trợ':'FG sẵn sàng tiếp nhận'}</h2><p>{admin?'Tin nhắn, ghi chú nội bộ và lịch sử xử lý theo quyền hiện tại.':'Gửi góp ý, báo lỗi hoặc bắt đầu chat riêng với quản trị.'}</p></div>}
  </div>
 </section>;
}
function NewSupport({client,onCreated,onCancel,onDenied}:{client:SupportClient;onCreated:(row:SupportUserRequest)=>void;onCancel:()=>void;onDenied:(error:unknown)=>void}){
 const [kind,setKind]=useState<SupportKind>('CHAT'),[title,setTitle]=useState(''),[body,setBody]=useState(''),[error,setError]=useState(''),[busy,setBusy]=useState(false),pending=useRef(false),controller=useRef<AbortController|null>(null),retry=useRef<{signature:string;id:string}|null>(null);
 useEffect(()=>()=>controller.current?.abort(),[]);
 async function submit(){if(pending.current)return;const signature=JSON.stringify({kind,title:kind==='CHAT'?undefined:title.trim(),body:body.trim()});if(retry.current?.signature!==signature)retry.current={signature,id:crypto.randomUUID()};const parsed=SupportCreateRequestSchema.safeParse({kind,clientRequestId:retry.current.id,...(kind==='CHAT'?{}:{title}),body});if(!parsed.success){setError('Vui lòng nhập nội dung hỗ trợ và chủ đề hợp lệ (tối đa 4.000 ký tự nội dung).');return;}pending.current=true;setBusy(true);setError('');const abort=new AbortController();controller.current=abort;try{const result=await client.create(parsed.data,abort.signal);if(!abort.signal.aborted)onCreated(result.request);}catch(problem){if(!abort.signal.aborted){setError(actionError(problem));if(problem instanceof ActionApiError&&[401,403].includes(problem.status))onDenied(problem);}}finally{pending.current=false;if(!abort.signal.aborted)setBusy(false);}}
 return <form className="fg-support-create" noValidate onSubmit={event=>{event.preventDefault();void submit();}}><h2>Yêu cầu hỗ trợ mới</h2><p>FG chỉ xác nhận tin đã được lưu. Quản trị sẽ phản hồi khi có thể; không yêu cầu mật khẩu của bạn.</p><label>Loại yêu cầu<select value={kind} disabled={busy} onChange={event=>setKind(event.target.value as SupportKind)}>{SupportKindSchema.options.map(value=><option key={value} value={value}>{kindLabel[value]}</option>)}</select></label>{kind!=='CHAT'&&<label>Chủ đề<input value={title} disabled={busy} maxLength={160} onChange={event=>setTitle(event.target.value)}/></label>}<label>Nội dung hỗ trợ<textarea value={body} disabled={busy} rows={7} onChange={event=>setBody(event.target.value)}/></label>{error&&<p role="alert" className="fg-action-error">{error}</p>}<footer><Button type="button" variant="outline" disabled={busy} onClick={onCancel}>Hủy</Button><Button type="submit" isLoading={busy}>Gửi yêu cầu</Button></footer></form>;
}
