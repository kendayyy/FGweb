import {useState} from 'react';import {Flag,Trash2} from 'lucide-react';import {useAdminAccess} from '../admin/AdminAccess.js';import {ActionDialog} from './ActionDialog.js';import {ReportDialog} from './ReportDialog.js';import {deleteMessage} from './api.js';import {ActionApiError} from './request.js';import {publishMessageRetraction} from './retractions.js';import './moderation.css';
interface Props{actorId?:string;conversationId:string;messageId:string;senderId:string;senderName:string;canManageRoom?:boolean;reportable?:boolean;onDeleted?:()=>void}
export function MessageActions(props:Props){if(!props.actorId)return null;return <OwnedActions key={props.actorId+':'+props.conversationId+':'+props.messageId} {...props} actorId={props.actorId}/>;}
function OwnedActions({actorId,conversationId,messageId,senderId,senderName,canManageRoom=false,reportable=true,onDeleted}:Props&{actorId:string}){
 const access=useAdminAccess(),[reporting,setReporting]=useState(false),[deleting,setDeleting]=useState(false),canDelete=actorId===senderId||canManageRoom||access.allowed&&access.actorId===actorId;
 return <><div className="fg-message-actions">{reportable&&<button type="button" aria-label={'Báo cáo tin nhắn của '+senderName} onClick={()=>setReporting(true)}><Flag size={15} aria-hidden="true"/>Báo cáo</button>}{canDelete&&<button type="button" aria-label={'Xóa tin nhắn của '+senderName} onClick={()=>setDeleting(true)}><Trash2 size={15} aria-hidden="true"/>Xóa tin</button>}</div>{reporting&&<ReportDialog target={{targetType:'MESSAGE',targetId:messageId,label:'Tin nhắn của '+senderName}} onClose={()=>setReporting(false)}/>} {deleting&&<ActionDialog title="Xóa tin nhắn" message="Tin và ảnh gắn với tin này sẽ không còn hiển thị trong hội thoại. Bản sao người nhận đã lưu trước đó không thể thu hồi từ thiết bị của họ." confirmLabel="Xác nhận xóa tin" onClose={()=>setDeleting(false)} onConfirm={async(reason,signal)=>{try{await deleteMessage(conversationId,messageId,reason,signal);if(!signal.aborted){publishMessageRetraction(conversationId,messageId);onDeleted?.();}}catch(error){if(error instanceof ActionApiError&&[401,403].includes(error.status)&&access.allowed)access.revoke();throw error;}} onSuccess={()=>setDeleting(false)}/>}</>;
}
export function ReportUserButton({actorId,targetId,label}:{actorId?:string;targetId:string;label:string}){if(!actorId||actorId===targetId)return null;return <OwnedUserReport key={actorId+':'+targetId} targetId={targetId} label={label}/>;}
function OwnedUserReport({targetId,label}:{targetId:string;label:string}){const [open,setOpen]=useState(false);return <><button type="button" className="fg-btn fg-btn-outline" onClick={()=>setOpen(true)}><Flag size={17} aria-hidden="true"/>Báo cáo người dùng</button>{open&&<ReportDialog target={{targetType:'USER',targetId,label}} onClose={()=>setOpen(false)}/>}</>;}
