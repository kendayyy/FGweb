
test('T38 actual legacy Lobby deletion retracts across two browser contexts',async({browser})=>{
  const runId='t38-delete-ui-'+randomUUID().replaceAll('-',''),dir='quality/evidence/T38/'+runId,startedAt=new Date().toISOString(),contexts:BrowserContext[]=[],captures:string[]=[];
  fs.mkdirSync(dir,{recursive:true});let fixture:Awaited<ReturnType<typeof startFixture>>|undefined,current:Page|undefined,step=0;
  async function capture(page:Page,title:string,result:string,status='RUNNING'){current=page;await page.bringToFront();const receipt=await captureUiStep(page,{task:'T38',runId,step:++step,title,action:title,result,startedAt,headed:true,privacySafe:true,status,maskSelectors:['[data-demo-private]','[data-sensitive]']});captures.push(receipt.id);}
  try{
    fixture=await startFixture(runId,dir);const {origin,actors,pool,games}=fixture;
    // Existing legacy compatibility source fixture; no new plaintext send, crypto downgrade,
    // custom encryption or E2EE claim. Current production history/deletion/socket paths run.
    const conversationId=games[0].lobby,source=randomUUID(),reply=randomUUID();
    await pool.query("INSERT INTO conversations(id,kind,lobby_id) VALUES($1,'LOBBY',$1)",[conversationId]);
    await pool.query('INSERT INTO messages(id,conversation_id,sender_id,content,client_message_id) VALUES($1,$2,$3,$4,$1)',[source,conversationId,actors.owner.id,'T38_LEGACY_SOURCE_RETRACTION']);
    await pool.query('INSERT INTO messages(id,conversation_id,sender_id,content,client_message_id,reply_to_id) VALUES($1,$2,$3,$4,$1,$5)',[reply,conversationId,actors.other.id,'T38_LEGACY_REPLY_REMAINS',source]);
    async function open(actor:'owner'|'other'){const context=await browser.newContext({viewport:{width:1440,height:900},reducedMotion:'reduce'});contexts.push(context);await context.addCookies([{name:'better-auth.session_token',value:actors[actor].cookie,url:origin,httpOnly:true,sameSite:'Lax'}]);const page=await context.newPage();page.setDefaultTimeout(10000);page.setDefaultNavigationTimeout(15000);await page.goto(origin+'/?game='+games[0].id,{waitUntil:'domcontentloaded'});await expect(page.locator('[data-message-id="'+source+'"]').getByText('T38_LEGACY_SOURCE_RETRACTION',{exact:true})).toBeVisible();await expect(page.locator('.fg-chat-status')).toContainText('Đã kết nối');return page;}
    const owner=await open('owner'),other=await open('other');current=owner;
    await expect(other.locator('[data-message-id="'+source+'"]').getByRole('button',{name:'Xóa tin nhắn của FG Test Owner'})).toHaveCount(0);
    await capture(other,'Bắt đầu: hai cửa sổ Lobby thật','Nguồn legacy tổng hợp và reply hiện qua API thật; người khác không có nút xóa của tác giả. Không claim dữ liệu này E2EE.');
    const sourceRow=owner.locator('[data-message-id="'+source+'"]');await owner.bringToFront();await sourceRow.getByRole('button',{name:'Trả lời tin nhắn của FG Test Owner'}).click();await sourceRow.getByRole('button',{name:'Xóa tin nhắn của FG Test Owner'}).click();let dialog=owner.getByRole('dialog',{name:'Xóa tin nhắn'});await dialog.getByRole('button',{name:'Hủy',exact:true}).click();expect((await pool.query('SELECT message_id FROM moderation_message_tombstones WHERE message_id=$1',[source])).rowCount).toBe(0);await expect(sourceRow).toBeVisible();await capture(owner,'Hủy xóa không ghi thay đổi','Hộp xác nhận đóng; source chưa bị tombstone và nội dung còn hiển thị.');
    await sourceRow.getByRole('button',{name:'Xóa tin nhắn của FG Test Owner'}).click();dialog=owner.getByRole('dialog',{name:'Xóa tin nhắn'});await dialog.getByRole('textbox',{name:'Lý do xử lý',exact:true}).fill('T38_HEADed_SOURCE_REASON');await dialog.getByRole('button',{name:'Xác nhận xóa tin',exact:true}).click();await expect(owner.locator('[data-message-id="'+source+'"]')).toHaveCount(0);await expect(other.locator('[data-message-id="'+source+'"]')).toHaveCount(0);await expect(other.locator('[data-message-id="'+reply+'"]')).toContainText('T38_LEGACY_REPLY_REMAINS');await expect(other.locator('.fg-reply-preview')).toHaveCount(0);await expect(owner.locator('.fg-composer-wrap .fg-reply-preview')).toHaveCount(0);
    const audit=(await pool.query('SELECT actor_id,reason FROM moderation_audit WHERE message_id=$1',[source])).rows;expect(audit).toEqual([{actor_id:actors.owner.id,reason:'T38_HEADed_SOURCE_REASON'}]);expect((await pool.query('SELECT id FROM messages WHERE id=$1',[source])).rowCount).toBe(1);await capture(other,'Socket thật thu hồi ở cửa sổ thứ hai','Tin gốc biến mất và trích dẫn bị bỏ, reply còn; server giữ source/audit và đúng lý do tác giả.');
    await other.reload({waitUntil:'domcontentloaded'});await expect(other.locator('[data-message-id="'+reply+'"]')).toBeVisible();await expect(other.getByText('T38_LEGACY_SOURCE_RETRACTION',{exact:true})).toHaveCount(0);await capture(other,'Kết thúc: tải lại không phục hồi nguồn đã xóa','History thật không đưa source hoặc reply preview trở lại; không đổi ciphertext hoặc đọc chat riêng.','SUCCEEDED');
    fs.writeFileSync(dir+'/result.json',JSON.stringify({runId,startedAt,completedAt:new Date().toISOString(),actualHeadedChrome:true,actualCompiledProductionApp:true,actualPostgresRedis:true,actualCrossContextSocketRetraction:true,legacyCompatibilityFixture:true,E2EEFixture:false,checks:['non-author control hidden','cancel no mutation','author reason confirmed','cross-context realtime source and reply preview removal','source and exact audit retained','fresh history excludes deleted source'],captureIds:captures,taskAcceptancePending:true},null,2)+'\n',{flag:'wx'});
  }catch(error){if(current&&!current.isClosed())await capture(current,'Lỗi kiểm thử thu hồi tin','Giữ lỗi thật; không chốt PASS task.','FAILED').catch(()=>{});fs.writeFileSync(dir+'/failure.json',JSON.stringify({at:new Date().toISOString(),captureIds:captures,taskAcceptancePending:true},null,2)+'\n',{flag:'wx'});throw error;}
  finally{try{for(const context of contexts)await context.close().catch(()=>{});}finally{await fixture?.close();}fs.writeFileSync(dir+'/browser-cleanup.json',JSON.stringify({at:new Date().toISOString(),ownedContextsClosed:true})+'\n',{flag:'wx'});}
});
