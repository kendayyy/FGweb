import {useCallback,useEffect,useRef,useState} from 'react';import {io,type Socket} from 'socket.io-client';import {Bot,ArrowLeft,Info,Lock} from 'lucide-react';import {z} from 'zod';import {SupportBodySchema,SupportAdminDetailSchema,SupportAdminRequestSchema,SupportStatusSchema,SupportIdSchema,SupportSocketEventNames as events,SupportMessageCreatedEventSchema,SupportUserRequestUpdatedEventSchema,SupportAdminRequestUpdatedEventSchema,SupportAdminNoteCreatedEventSchema,SUPPORT_RECEIPT_TEXT,type SupportMessage,type SupportNote,type SupportAudit,type SupportStatus} from '@fg/contracts';import {Button} from '../../components/ui/Button.js';import {InitialLoading} from '../../components/ui/InitialLoading.js';import {registerAuthCleanup} from '../auth/lifecycle.js';import {useAdminAccess} from '../admin/AdminAccess.js';import {ActionApiError,actionError} from '../moderation/request.js';import {SupportClient} from './api.js';import {emptyDraft,retryDraft,type SupportDraft} from './drafts.js';import {statusLabel,kindLabel,supportTime,type SupportRequest} from './labels.js';
type Detail=Awaited<ReturnType<SupportClient['detail']>>;
const merge=<T extends{id:string}>(old:T[],added:T[])=>{const rows=new Map(old.map(row=>[row.id,row]));for(const row of added)rows.set(row.id,row);return[...rows.values()];};
interface Props{id:string;client:SupportClient;vault:Map<string,SupportDraft>;showThread:boolean;showDetails:boolean;compact:boolean;onBack:()=>void;onDetails:()=>void;onThread:()=>void;onUpdated:(row:SupportRequest)=>void;onDenied:(error:unknown)=>void}
export function SupportThread({id,client,vault,showThread,showDetails,compact,onBack,onDetails,onThread,onUpdated,onDenied}:Props){
 const access=useAdminAccess(),[detail,setDetail]=useState<Detail|null>(null),[messages,setMessages]=useState<SupportMessage[]>([]),[notes,setNotes]=useState<SupportNote[]>([]),[audit,setAudit]=useState<SupportAudit[]>([]),[cursor,setCursor]=useState<number|null>(null),[noteCursor,setNoteCursor]=useState<string|null>(null),[auditCursor,setAuditCursor]=useState<string|null>(null),[loading,setLoading]=useState(true),[error,setError]=useState(''),[actionProblem,setActionProblem]=useState(''),[online,setOnline]=useState(false),[busy,setBusy]=useState<'message'|'note'|'read'|'status'|'more'|'notes'|'audit'|null>(null),[message,setMessage]=useState(()=>vault.get(id)?.message??''),[note,setNote]=useState(()=>vault.get(id)?.note??''),[receipt,setReceipt]=useState('');
 const active=useRef(true),pending=useRef(false),controllers=useRef(new Set<AbortController>()),latest=useRef<AbortController|null>(null),socket=useRef<Socket|null>(null),seenNotes=useRef(new Set<string>()),seenAudit=useRef(new Set<string>());
 const draft=()=>{let row=vault.get(id);if(!row){row=emptyDraft();vault.set(id,row);}return row;};
 const canAct=online&&(!client.admin||access.online);
 const failure=useCallback((problem:unknown)=>{if(!active.current)return;setError(actionError(problem));if(problem instanceof ActionApiError&&[401,403,404].includes(problem.status)){setDetail(null);setMessages([]);setNotes([]);setAudit([]);vault.delete(id);onDenied(problem);}},[id,vault,onDenied]);
 const refresh=useCallback(async()=>{if(!active.current)return;latest.current?.abort();const controller=new AbortController();latest.current=controller;controllers.current.add(controller);try{const [next,page]=await Promise.all([client.detail(id,controller.signal),client.messages(id,controller.signal)]);if(controller.signal.aborted||!active.current)return;setDetail(next);onUpdated(next.request);setMessages(previous=>merge(previous,page.items).sort((a,b)=>a.sequence-b.sequence));setCursor(page.nextAfterSequence);const internal=client.admin?SupportAdminDetailSchema.parse(next):undefined;if(internal){setNotes(previous=>merge(previous,internal.notes.items).sort((a,b)=>b.createdAt.localeCompare(a.createdAt)||b.id.localeCompare(a.id)));setAudit(previous=>merge(previous,internal.audit.items).sort((a,b)=>b.createdAt.localeCompare(a.createdAt)||b.id.localeCompare(a.id)));setNoteCursor(previous=>seenNotes.current.size?previous:internal.notes.nextCursor);setAuditCursor(previous=>seenAudit.current.size?previous:internal.audit.nextCursor);}
    const current=vault.get(id);if(current?.pendingMessage&&page.items.some(row=>row.authorId===client.actorId&&row.clientMessageId===current.pendingMessage!.id&&row.body===current.pendingMessage!.body)&&current.message.trim()===current.pendingMessage.body){current.message='';current.pendingMessage=undefined;setMessage('');setReceipt(SUPPORT_RECEIPT_TEXT);}if(current?.pendingNote&&internal&&internal.notes.items.some(row=>row.authorId===client.actorId&&row.clientNoteId===current.pendingNote!.id&&row.body===current.pendingNote!.body)&&current.note.trim()===current.pendingNote.body){current.note='';current.pendingNote=undefined;setNote('');}
    setError('');setLoading(false);
   }catch(problem){if(!controller.signal.aborted&&active.current){setLoading(false);failure(problem);}}finally{controllers.current.delete(controller);}},[client,id,onUpdated,failure,vault]);
 useEffect(()=>{active.current=true;void refresh();const connection=io('/support',{transports:['websocket'],withCredentials:true,autoConnect:false,reconnection:true,forceNew:true,timeout:5000});socket.current=connection;let busyRefresh=false,queued=false;
  const wake=async()=>{if(!active.current)return;if(busyRefresh){queued=true;return;}busyRefresh=true;try{await refresh();}finally{busyRefresh=false;if(queued&&active.current){queued=false;void wake();}}};
  connection.on('connect',()=>{void(async()=>{try{const raw=await connection.timeout(5000).emitWithAck(events.SUBSCRIBE,{requestId:id}),ack=z.object({ok:z.literal(true),requestId:SupportIdSchema}).strict().safeParse(raw);if(!active.current)return;if(!ack.success||ack.data.requestId!==id){failure(new ActionApiError(403));return;}setOnline(true);void wake();}catch{if(active.current)setOnline(false);}})();});
  connection.on('disconnect',(reason:string)=>{if(!active.current)return;setOnline(false);if(reason==='io server disconnect')failure(new ActionApiError(403));});connection.on('connect_error',()=>{if(active.current)setOnline(false);});
  connection.on(events.MESSAGE_CREATED,(value:unknown)=>{const parsed=SupportMessageCreatedEventSchema.safeParse(value);if(parsed.success&&parsed.data.requestId===id)void wake();});
  connection.on(client.admin?events.ADMIN_REQUEST_UPDATED:events.REQUEST_UPDATED,(value:unknown)=>{const parsed=client.admin?SupportAdminRequestUpdatedEventSchema.safeParse(value):SupportUserRequestUpdatedEventSchema.safeParse(value);if(parsed.success&&parsed.data.request.id===id&&(client.admin||parsed.data.request.ownerId===client.actorId))void wake();});
  if(client.admin)connection.on(events.ADMIN_NOTE_CREATED,(value:unknown)=>{const parsed=SupportAdminNoteCreatedEventSchema.safeParse(value);if(parsed.success&&parsed.data.requestId===id)void wake();});connection.connect();
  const timer=setInterval(()=>{if(!document.hidden)void wake();},15000),visible=()=>{if(!document.hidden)void wake();};document.addEventListener('visibilitychange',visible);
  const stop=()=>{active.current=false;for(const controller of controllers.current)controller.abort();connection.removeAllListeners();connection.disconnect();setDetail(null);setMessages([]);setNotes([]);setAudit([]);setMessage('');setNote('');vault.delete(id);};const unregister=registerAuthCleanup(stop);
  return()=>{unregister();active.current=false;for(const controller of controllers.current)controller.abort();clearInterval(timer);document.removeEventListener('visibilitychange',visible);connection.removeAllListeners();connection.disconnect();};
 },[client,id,refresh,failure,vault]);
 async function perform(kind:NonNullable<typeof busy>,work:(signal:AbortSignal)=>Promise<void>){if(pending.current||!canAct||!active.current)return;pending.current=true;setBusy(kind);setActionProblem('');const controller=new AbortController();controllers.current.add(controller);try{await work(controller.signal);}catch(problem){if(!controller.signal.aborted&&active.current){setActionProblem(actionError(problem));if(problem instanceof ActionApiError&&[401,403,404].includes(problem.status))failure(problem);if(problem instanceof ActionApiError&&problem.status===409)void refresh();}}finally{pending.current=false;controllers.current.delete(controller);if(!controller.signal.aborted&&active.current)setBusy(null);}}
 function send(){const parsed=SupportBodySchema.safeParse(message);if(!parsed.success){setActionProblem('Vui lòng nhập nội dung tin nhắn, tối đa 4.000 ký tự.');return;}const current=draft(),retry=retryDraft(current.pendingMessage,parsed.data);current.pendingMessage=retry;void perform('message',async signal=>{const result=await client.send(id,{clientMessageId:retry.id,body:retry.body},signal);if(signal.aborted||!active.current)return;current.message='';current.pendingMessage=undefined;setMessage('');setReceipt(result.receipt.text);setMessages(previous=>merge(previous,[result.message]).sort((a,b)=>a.sequence-b.sequence));await refresh();});}
 function saveNote(){const parsed=SupportBodySchema.safeParse(note);if(!parsed.success){setActionProblem('Vui lòng nhập nội dung ghi chú, tối đa 4.000 ký tự.');return;}const current=draft(),retry=retryDraft(current.pendingNote,parsed.data);current.pendingNote=retry;void perform('note',async signal=>{const result=await client.note(id,{clientNoteId:retry.id,body:retry.body},signal);if(signal.aborted||!active.current)return;current.note='';current.pendingNote=undefined;setNote('');setNotes(previous=>merge(previous,[result]));setReceipt('Đã lưu ghi chú nội bộ.');await refresh();});}
 const request=detail?.request,adminRequest=SupportAdminRequestSchema.safeParse(request),ownerLabel=adminRequest.success?adminRequest.data.owner.displayName:'Người gửi';
 return <>{showThread&&<section className="fg-support-thread" aria-label="Cuộc hỗ trợ">{compact&&<div className="fg-support-mobile-tools"><Button variant="ghost" onClick={onBack} leftIcon={<ArrowLeft size={18} aria-hidden="true"/>}>Hộp thư</Button>{client.admin&&<Button variant="outline" onClick={onDetails}>Chi tiết yêu cầu</Button>}</div>}<header className="fg-support-thread-heading"><div><h2>{request?.title??'Cuộc hỗ trợ'}</h2>{request&&<span className={'fg-support-status '+request.status}>{statusLabel[request.status]}</span>}</div><div><span className="fg-support-connection">{online?'Đã kết nối':'Mất kết nối · đang kiểm tra lại'}</span><Button variant="outline" onClick={()=>void refresh()}>Tải lại cuộc hỗ trợ</Button></div></header>
  {loading&&!detail?<InitialLoading pending layout="chat" label="Đang tải cuộc hỗ trợ riêng"/>:error&&!detail?<p role="alert" className="fg-action-error">{error}</p>:<><div className="fg-support-messages" aria-label="Lịch sử hỗ trợ">{messages.map(row=><article key={row.id} className={'fg-support-message '+(row.authorRole==='SYSTEM'?'is-system':row.authorId===client.actorId?'is-own':'')} data-message-id={row.id}>{row.authorRole==='SYSTEM'?<Bot size={18} aria-hidden="true"/>:<strong>{row.authorId===client.actorId?'Bạn':row.authorRole==='ADMIN'?'Quản trị FG':ownerLabel}</strong>}<p>{row.body}</p><time dateTime={row.createdAt}>{supportTime(row.createdAt)}</time></article>)}</div>{cursor!==null&&<Button variant="outline" disabled={Boolean(busy)||!canAct} onClick={()=>void perform('more',async signal=>{const page=await client.messages(id,signal,cursor);if(signal.aborted||!active.current)return;setMessages(previous=>merge(previous,page.items).sort((a,b)=>a.sequence-b.sequence));setCursor(page.nextAfterSequence);})}>Tải thêm tin nhắn</Button>}<div className="fg-support-read"><span>{request?.unreadCount??0} tin chưa đọc theo server</span><Button variant="ghost" disabled={!messages.length||Boolean(busy)||!canAct} onClick={()=>void perform('read',async signal=>{await client.read(id,{lastReadMessageId:messages.at(-1)!.id},signal);if(!signal.aborted)await refresh();})}>Đánh dấu đã đọc</Button></div><form className="fg-support-composer" noValidate onSubmit={event=>{event.preventDefault();send();}}><label>Tin nhắn hỗ trợ<textarea rows={3} value={message} disabled={Boolean(busy)} onChange={event=>{draft().message=event.target.value;setMessage(event.target.value);}}/></label>{actionProblem&&<p role="alert" className="fg-action-error">{actionProblem}</p>}{receipt&&<p role="status" className="fg-support-receipt">{receipt}</p>}<footer><small>Không gửi mật khẩu hoặc mã khôi phục.</small><Button type="submit" isLoading={busy==='message'} disabled={Boolean(busy)||!canAct}>{client.admin?'Gửi phản hồi':'Gửi tin nhắn'}</Button></footer></form></>}
 </section>}
 {showDetails&&client.admin&&<aside className="fg-support-detail" aria-label="Chi tiết và ghi chú nội bộ">{compact&&<Button variant="ghost" onClick={onThread} leftIcon={<ArrowLeft size={18} aria-hidden="true"/>}>Cuộc hỗ trợ</Button>}<h2>Chi tiết yêu cầu</h2>{request?<><dl><dt>Mã yêu cầu</dt><dd><code>{request.id}</code></dd><dt>Người gửi</dt><dd>{ownerLabel}</dd><dt>Loại yêu cầu</dt><dd>{kindLabel[request.kind]}</dd><dt>Tiếp nhận lúc</dt><dd><time dateTime={request.createdAt}>{supportTime(request.createdAt)}</time></dd></dl><label>Trạng thái yêu cầu<select value={request.status} disabled={Boolean(busy)||!canAct} onChange={event=>{const next=event.target.value as SupportStatus;void perform('status',async signal=>{const result=await client.status(id,{status:next,expectedVersion:request.version},signal);if(signal.aborted||!active.current)return;setDetail(previous=>previous?{...previous,request:result}:previous);onUpdated(result);setReceipt('Đã cập nhật trạng thái yêu cầu.');await refresh();});}}>{SupportStatusSchema.options.map(value=><option key={value} value={value}>{statusLabel[value]}</option>)}</select></label><section className="fg-support-internal"><h3><Lock size={17} aria-hidden="true"/>Ghi chú nội bộ</h3><p>Chỉ quản trị xem</p><label>Nội dung ghi chú nội bộ<textarea value={note} disabled={Boolean(busy)} rows={4} onChange={event=>{draft().note=event.target.value;setNote(event.target.value);}}/></label><Button variant="outline" disabled={Boolean(busy)||!canAct} isLoading={busy==='note'} onClick={saveNote}>Lưu ghi chú</Button>{notes.map(row=><article key={row.id}><p>{row.body}</p><time dateTime={row.createdAt}>{supportTime(row.createdAt)}</time></article>)}{noteCursor&&<Button variant="ghost" disabled={Boolean(busy)||!canAct} onClick={()=>void perform('notes',async signal=>{if(seenNotes.current.has(noteCursor))throw new ActionApiError(0,true);const page=await client.notes(id,signal,noteCursor);if(signal.aborted||!active.current)return;seenNotes.current.add(noteCursor);setNotes(previous=>merge(previous,page.items));setNoteCursor(page.nextCursor);})}>Tải thêm ghi chú</Button>}</section><section className="fg-support-audit"><h3><Info size={17} aria-hidden="true"/>Lịch sử xử lý</h3><ol>{audit.map(row=><li key={row.id}><time dateTime={row.createdAt}>{supportTime(row.createdAt)}</time><span>{row.action==='CREATED'?'Tạo yêu cầu':row.action==='NOTE_ADDED'?'Thêm ghi chú nội bộ':'Chuyển sang '+statusLabel[row.toStatus!]}</span></li>)}</ol>{auditCursor&&<Button variant="ghost" disabled={Boolean(busy)||!canAct} onClick={()=>void perform('audit',async signal=>{if(seenAudit.current.has(auditCursor))throw new ActionApiError(0,true);const page=await client.audit(id,signal,auditCursor);if(signal.aborted||!active.current)return;seenAudit.current.add(auditCursor);setAudit(previous=>merge(previous,page.items));setAuditCursor(page.nextCursor);})}>Tải thêm lịch sử xử lý</Button>}</section>{actionProblem&&(!showThread||compact)&&<p role="alert" className="fg-action-error">{actionProblem}</p>}</>:<p role="status">Đang tải chi tiết…</p>}</aside>}
 </>;
}
