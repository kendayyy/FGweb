import {useEffect,useState} from 'react';import type {z} from 'zod';import type {DisclosedReportSchema} from '@fg/contracts';import {Button} from '../../components/ui/Button.js';import {ActionDialog} from '../moderation/ActionDialog.js';import {deleteMessage} from '../moderation/api.js';import {ActionApiError,actionError} from '../moderation/request.js';import {publishMessageRetraction} from '../moderation/retractions.js';import {useAdminAccess} from './AdminAccess.js';import {readDisclosedReport,disclosedImage} from './api.js';
type Evidence=z.infer<typeof DisclosedReportSchema>;
export function DisclosedEvidence({reportId}:{reportId:string}){
 const access=useAdminAccess(),[data,setData]=useState<Evidence|null>(null),[error,setError]=useState(''),[loading,setLoading]=useState(true),[image,setImage]=useState<string|null>(null),[deleting,setDeleting]=useState<{conversationId:string;messageId:string}|null>(null),[removed,setRemoved]=useState(new Set<string>());
 useEffect(()=>{const controller=new AbortController();let alive=true;setData(null);setError('');setLoading(true);void readDisclosedReport(reportId,controller.signal).then(value=>{if(alive&&!controller.signal.aborted){setData(value);setLoading(false);}}).catch(problem=>{if(!alive||controller.signal.aborted)return;setLoading(false);if(problem instanceof ActionApiError&&[401,403].includes(problem.status)){access.revoke();return;}setError(problem instanceof ActionApiError&&problem.status===404?'Báo cáo không có bằng chứng được chia sẻ hoặc bằng chứng không còn khả dụng.':actionError(problem));});return()=>{alive=false;controller.abort();};},[reportId,access.revoke]);
 const allowed=data?[...new Set(data.selectedMessages.flatMap(row=>row.reportAttachmentIds))]:[];
 return <section className="fg-admin-evidence" aria-label="Bằng chứng được chia sẻ"><h3>Bằng chứng người dùng chủ động chia sẻ</h3><p className="fg-admin-warning">Nội dung do người dùng cung cấp; chưa được xác minh tính xác thực. Đây không phải quyền đọc toàn bộ chat riêng.</p>{loading&&<p role="status">Đang đọc bằng chứng theo quyền hiện tại…</p>}{error&&<p role="alert" className="fg-action-error">{error}</p>}{data?.selectedMessages.map((row,index)=><article key={row.evidenceId}><h4>Tin được chọn {index+1}</h4><p className="fg-admin-evidence-text">{row.text||'Tin có ảnh được người dùng chọn'}</p><small>Tham chiếu tin: <code>{row.messageId}</code></small><div className="fg-admin-row-actions">{row.reportAttachmentIds.map((id,imageIndex)=><Button key={id} variant="outline" disabled={!access.online} onClick={()=>setImage(id)}>Xem ảnh bằng chứng {imageIndex+1}</Button>)}{removed.has(row.messageId)?<p role="status">Tin gốc đã được xóa khỏi hội thoại; bằng chứng đã chia sẻ vẫn được giữ theo audit.</p>:<Button variant="destructive" disabled={!access.online} onClick={()=>setDeleting({conversationId:row.conversationId,messageId:row.messageId})}>Xóa tin gốc đã báo cáo</Button>}</div></article>)}{image&&<div className="fg-admin-evidence-image"><Button variant="ghost" onClick={()=>setImage(null)}>Đóng ảnh bằng chứng</Button><EvidenceImage key={reportId+':'+image} reportId={reportId} attachmentId={image} allowed={allowed}/></div>}{deleting&&<ActionDialog title="Xóa tin nhắn" message="Xóa tin gốc và ảnh đính kèm khỏi hội thoại. Bằng chứng đã được người dùng chia sẻ vẫn được giữ riêng theo audit; nội dung chat khác không được mở." confirmLabel="Xác nhận xóa tin" onClose={()=>setDeleting(null)} onConfirm={async(reason,signal)=>{if(!access.online)throw new ActionApiError(0);try{await deleteMessage(deleting.conversationId,deleting.messageId,reason,signal);if(!signal.aborted){publishMessageRetraction(deleting.conversationId,deleting.messageId);setRemoved(previous=>new Set([...previous,deleting.messageId]));}}catch(problem){if(problem instanceof ActionApiError&&[401,403].includes(problem.status))access.revoke();throw problem;}} onSuccess={()=>setDeleting(null)}/>}</section>;
}
function EvidenceImage({reportId,attachmentId,allowed}:{reportId:string;attachmentId:string;allowed:string[]}){const access=useAdminAccess(),[url,setUrl]=useState<string|null>(null),[error,setError]=useState('');useEffect(()=>{const controller=new AbortController();let owned:string|undefined;void disclosedImage(reportId,attachmentId,allowed,controller.signal).then(blob=>{if(controller.signal.aborted)return;owned=URL.createObjectURL(blob);setUrl(owned);}).catch(problem=>{if(controller.signal.aborted)return;if(problem instanceof ActionApiError&&[401,403].includes(problem.status))access.revoke();else setError(actionError(problem));});return()=>{controller.abort();if(owned)URL.revokeObjectURL(owned);};},[reportId,attachmentId,access.revoke]);return error?<p role="alert" className="fg-action-error">{error}</p>:url?<img src={url} alt="Ảnh bằng chứng được người dùng chia sẻ"/>:<p role="status">Đang đọc ảnh bằng chứng…</p>;}
